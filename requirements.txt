for (i,dom,lab,var,cod,lvl,role,s,tier) in L:
    s=pd.Series(s,index=d.index).astype(float)
    ok=s.notna()
    n=int(ok.sum())
    if n<500: rows.append(dict(id=i,domain=dom,label=lab,vars=var,coding=cod,level=lvl,role=role,tier=tier,n=n,prev=None,icc=None,inf_psu=None,inf_child=None,binary=None)); continue
    x=s[ok]; ww=pd.Series(w,index=d.index)[ok]; g=d.PSU[ok]
    binary=set(np.unique(x.dropna()))<= {0.0,1.0}
    prev=float(np.average(x,weights=ww)) if binary else float(np.average(x,weights=ww))
    df=pd.DataFrame({'x':x,'g':g})
    gs=df.groupby('g').x.agg(['count','mean','sum'])
    N=len(df);k=len(gs)
    ssb=(gs['count']*(gs['mean']-df.x.mean())**2).sum()
    ssw=((df.x-df.groupby('g').x.transform('mean'))**2).sum()
    msb=ssb/(k-1); msw=ssw/(N-k)
    n0=(N-(gs['count']**2).sum()/N)/(k-1)
    icc=(msb-msw)/(msb+(n0-1)*msw)
    if binary:
        gg=df.groupby('g').x.agg(['count','min','max'])
        inf=gg[(gg['count']>=2)&(gg['min']!=gg['max'])]
        inf_psu=int(len(inf)); inf_child=int(inf['count'].sum())
    else:
        inf_psu=int((gs['count']>=2).sum()); inf_child=int(gs.loc[gs['count']>=2,'count'].sum())
    rows.append(dict(id=i,domain=dom,label=lab,vars=var,coding=cod,level=lvl,role=role,tier=tier,n=n,prev=prev,icc=float(icc),inf_psu=inf_psu,inf_child=inf_child,binary=bool(binary)))
